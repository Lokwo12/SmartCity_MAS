:- dynamic bin_level/1, bin_state/1, hb_counter/1, started/0, last_reset_req/1.
:- dynamic last_monitor_ts/1.

bin_level(0).
max_capacity(100).
fill_step(20).
bin_state(idle).
shared_token(city_token_2026).
deltaT(2).
monitor_interval_ms(1000).

short_bin_id(Bin,Short) :-
    atom_concat(smart_bin,N,Bin),
    !,
    atom_concat(sb,N,Short).
short_bin_id(Bin,Bin).

startI :>
    retractall(started),
    assert(started),
    agent(BinId),
    write('[SMARTBIN '), write(BinId), write('] startI fired'), nl,
    deltaT(D),
    retractall(hb_counter(_)),
    retractall(last_reset_req(_)),
    retractall(last_monitor_ts(_)),
    assert(hb_counter(D)),
    statistics(walltime,[Now,_]),
    assert(last_monitor_ts(Now)),
    format('[BIN ~w] Online | Level=0% | Monitoring...~n',[BinId]),
    true.

monitorI(dummy) :>
    statistics(walltime,[Now,_]),
    last_monitor_ts(Last),
    monitor_interval_ms(Interval),
    Now - Last < Interval,
    !,
    true.

monitorI(dummy) :>
    statistics(walltime,[Now,_]),
    last_monitor_ts(Last),
    monitor_interval_ms(Interval),
    Now - Last >= Interval,
    retract(last_monitor_ts(Last)),
    assert(last_monitor_ts(Now)),
    monitor_tickA.

monitor_tickA :-
    hb_counter(C),
    C > 1,
    C1 is C - 1,
    retract(hb_counter(C)),
    assert(hb_counter(C1)).

monitor_tickA :-
    hb_counter(1),
    deltaT(D),
    retract(hb_counter(1)),
    assert(hb_counter(D)),
    tickI.

monitor_tickA :- true.

tickI :>
    bin_state(waiting),
    true.

tickI :>
    bin_state(idle),
    maybe_fillI.

maybe_fillI :>
    bin_state(idle),
    bin_level(L),
    max_capacity(M),
    L >= M,
    !,
    evi(full_trigger).

maybe_fillI :>
    bin_state(idle),
    increase_levelA.

increase_levelA :-
    bin_level(L),
    max_capacity(M),
    L < M,
    fill_step(S),
    NL is min(L+S, M),
    retract(bin_level(L)),
    assert(bin_level(NL)),
    agent(BinId),
    format('[BIN ~w] Level: ~w% -> ~w%~n',[BinId,L,NL]),
    messageA(logger, log(info,level_update,BinId)),
    ( NL >= M -> evi(full_trigger) ; true ).

full_triggerI :>
    retract(bin_state(idle)),
    assert(bin_state(waiting)),
    agent(BinId),
    format('~n[BIN ~w] *** FULL (100%) *** Sending alert to Control Center~n',[BinId]),
    shared_token(Token),
    messageA(control_center, bin_full(BinId,Token)),
    messageA(logger, log(info,bin_full,BinId)).

reset_binE(Bin,ReqId,Token) :>
    shared_token(Token),
    agent(Bin),
    \+ last_reset_req(ReqId),
    !,
    assert(last_reset_req(ReqId)),
    apply_reset_nowA,
    messageA(control_center, reset_ack(Bin,ReqId,Token)).

reset_binE(Bin,ReqId,Token) :>
    shared_token(Token),
    agent(Bin),
    last_reset_req(ReqId),
    !.

apply_reset_nowA :-
    retractall(bin_level(_)), assert(bin_level(0)),
    retractall(bin_state(_)), assert(bin_state(idle)),
    agent(BinId),
    format('[BIN ~w] EMPTIED by truck | Level reset to 0% | Resuming fill~n',[BinId]),
    messageA(logger, log(info,bin_reset,BinId)).

retry_collectionE(Bin,ReqId,Token) :>
    shared_token(Token),
    agent(Bin),
    shared_token(Tok),
    messageA(control_center, bin_full(Bin,Tok)).









fireEventA(E) :-
    ( catch(call(eve(E)),_,fail)
    ; catch(call(evi(E)),_,fail)
    ), !.
fireEventA(_).

fire_event(E) :- fireEventA(E).

% send_message handlers - strip Token, fire simple-arity eve() (proven DALI pattern).
reset_bin(Bin,ReqId,Token) :-
    shared_token(Token), agent(Bin), fire_event(reset_bin(Bin,ReqId,Token)).
retry_collection(Bin,ReqId,Token) :-
    shared_token(Token), agent(Bin), fire_event(retry_collection(Bin,ReqId,Token)).
send_message(reset_bin(Bin,ReqId,Token),_From) :-
    shared_token(Token), agent(Bin), fire_event(reset_bin(Bin,ReqId,Token)).
send_message(retry_collection(Bin,ReqId,Token),_From) :-
    shared_token(Token), agent(Bin), fire_event(retry_collection(Bin,ReqId,Token)).

monitor(dummy) :-
    evi(monitor(dummy)).
