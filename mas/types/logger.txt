:- dynamic log_store/4, started/0, recent_log/2.

dedup_window_ms(1200).

log_printI(Level,Sender,Msg) :>
    statistics(walltime,[T,_]),
    ( Level = info -> Lvl = 'INFO'
    ; Level = warn -> Lvl = 'WARN'
    ; Level = error -> Lvl = 'ERROR'
    ; Lvl = Level
    ),
    format('[logger +~wms] ~w src=~w | ~w~n',[T,Lvl,Sender,Msg]).

startI :>
    retractall(started),
    assert(started),
    retractall(log_store(_,_,_,_)),
    retractall(recent_log(_,_)).

should_printI(Key,_Level,_Sender,_Msg) :>
    statistics(walltime,[Now,_]),
    dedup_window_ms(W),
    recent_log(Key,Prev),
    Now - Prev < W,
    !,
    true.

should_printI(Key,Level,Sender,Msg) :>
    statistics(walltime,[Now,_]),
    retractall(recent_log(Key,_)),
    assert(recent_log(Key,Now)),
    evi(log_print(Level,Sender,Msg)).

logI(log(Level,Event,Sender)) :>
    statistics(walltime, [T,_]),
    assert(log_store(T,Sender,Event,Level)),
    evi(should_print(k(Level,Sender,Event),Level,Sender,Event)).

logI(log(Level,event(Name,req(ReqId),bin(Bin),truck(Truck),note(Note)),Sender)) :>
    statistics(walltime, [T,_]),
    assert(log_store(T,Sender,event(Name,req(ReqId),bin(Bin),truck(Truck),note(Note)),Level)),
    ( atom_concat(smart_bin,N,Bin) -> atom_concat(smartbin,N,Bp) ; Bp = Bin ),
    format(atom(Msg),'~w bin=~w truck=~w note=~w',[Name,Bp,Truck,Note]),
    evi(should_print(k(Level,Sender,event(Name,ReqId,Bin,Truck,Note)),Level,Sender,Msg)).

logI(log(error,Event,Sender)) :>
    evi(should_print(k(error,Sender,Event),error,Sender,Event)).

send_message(send_message(Msg,_),From) :-
    !,
    send_message(Msg,From).

send_message(inform(Msg,_),From) :-
    !,
    send_message(Msg,From).

send_message(Msg,From) :-
    evi(log_in(Msg,From)).

log_inI(inform(log(Level,Event,Sender),_),_From) :>
    evi(log(log(Level,Event,Sender))).

log_inI(log(Level,Event,Sender),_From) :>
    evi(log(log(Level,Event,Sender))).

log_inI(send_message(log(Level,Event,Sender),_),_From) :>
    evi(log(log(Level,Event,Sender))).

log_inI(send_message(inform(log(Level,Event,Sender),_),_),_From) :>
    evi(log(log(Level,Event,Sender))).

monitorI(dummy) :>
    started.

monitorI(dummy) :>
    \+ started,
    evi(start).

monitor(dummy) :-
    evi(monitor(dummy)).
