:- dynamic bin_level/1, bin_state/1, hb_counter/1, started/0, last_reset_req/1, last_monitor_ts/1.

bin_level(0).
max_capacity(100).
fill_step(20).
bin_state(idle).
shared_token(city_token_2026).
deltaT(2).
monitor_interval_ms(1000).

send_level_log_traceA(BinId) :-
    messageA(logger, send_message(log(info,level_update,BinId),BinId), BinId),
    !,
    format('[SMART_BIN ~w] SEND_OK logger level_update~n',[BinId]).
send_level_log_traceA(BinId) :-
    format('[SMART_BIN ~w] SEND_FAIL logger level_update~n',[BinId]).

send_full_traceA(BinId,Token) :-
    messageA(control_center, send_message(bin_full(BinId,Token),BinId), BinId),
    !,
    format('[SMART_BIN ~w] SEND_OK control_center bin_full~n',[BinId]).
send_full_traceA(BinId,_Token) :-
    format('[SMART_BIN ~w] SEND_FAIL control_center bin_full~n',[BinId]).

send_bin_full_log_traceA(BinId) :-
    messageA(logger, send_message(log(info,bin_full,BinId),BinId), BinId),
    !,
    format('[SMART_BIN ~w] SEND_OK logger bin_full~n',[BinId]).
send_bin_full_log_traceA(BinId) :-
    format('[SMART_BIN ~w] SEND_FAIL logger bin_full~n',[BinId]).

startI :>
    retractall(started),
    assert(started),
    agent(BinId),
    write('[SMARTBIN '), write(BinId), write('] startI fired'), nl,
    deltaT(D),
    retractall(hb_counter(_)),
    retractall(last_reset_req(_)),
    retractall(last_monitor_ts(_)),
    assert(hb_counter(D)),
    statistics(walltime,[Now,_]),
    assert(last_monitor_ts(Now)),
    format('[SMART_BIN ~w] START~n',[BinId]),
    true.

monitorI(dummy) :>
    statistics(walltime,[Now,_]),
    last_monitor_ts(Last),
    monitor_interval_ms(Interval),
    Now - Last < Interval,
    !,
    true.

monitorI(dummy) :>
    statistics(walltime,[Now,_]),
    last_monitor_ts(Last),
    monitor_interval_ms(Interval),
    Now - Last >= Interval,
    retract(last_monitor_ts(Last)),
    assert(last_monitor_ts(Now)),
    evi(monitor_tick).

monitor_tickI :>
    hb_counter(C),
    C > 1,
    C1 is C - 1,
    retract(hb_counter(C)),
    assert(hb_counter(C1)).

monitor_tickI :>
    hb_counter(1),
    deltaT(D),
    retract(hb_counter(1)),
    assert(hb_counter(D)),
    evi(tick).

monitor_tickI :> true.

tickI :>
    bin_state(waiting),
    true.

tickI :>
    bin_state(idle),
    maybe_fillI.

maybe_fillI :>
    bin_state(idle),
    evi(increase_level).

increase_levelI :>
    bin_level(L),
    max_capacity(M),
    L < M,
    fill_step(S),
    NL is min(L+S, M),
    retract(bin_level(L)),
    assert(bin_level(NL)),
    agent(BinId),
    format('[SMART_BIN ~w] LEVEL=~w%~n',[BinId,NL]),
        send_level_log_traceA(BinId),
    ( NL >= M -> evi(full_trigger) ; true ).

full_triggerI :>
    bin_state(idle),
    retract(bin_state(idle)),
    assert(bin_state(waiting)),
    agent(BinId),
    format('[SMART_BIN ~w] FULL -> notify control_center~n',[BinId]),
    shared_token(Token),
        send_full_traceA(BinId,Token),
        send_bin_full_log_traceA(BinId).

reset_binI :>
    evi(apply_reset_now).

apply_reset_nowI :>
    retractall(bin_level(_)), assert(bin_level(0)),
    retractall(bin_state(_)), assert(bin_state(idle)),
    agent(BinId),
    format('[SMART_BIN ~w] RESET -> level=0%~n',[BinId]),
    messageA(logger, send_message(log(info,bin_reset,BinId),BinId), BinId).

retry_collectionI :>
    agent(BinId),
    shared_token(Token),
    messageA(control_center, send_message(bin_full(BinId,Token),BinId), BinId).

send_message(send_message(Msg,_),From) :-
    !,
    send_message(Msg,From).

send_message(inform(Msg,_),From) :-
    !,
    send_message(Msg,From).

send_message(Msg,From) :-
    evi(bin_in(Msg,From)).

bin_inI(bin_full(Bin,Token),_From) :>
    shared_token(Token),
    evi(bin_full(Bin)).

bin_inI(reset_bin,_From) :>
    evi(reset_bin).

bin_inI(reset_bin(Bin,ReqId,Token),control_center) :>
    shared_token(Token),
    agent(Bin),
    last_reset_req(ReqId),
    messageA(control_center, send_message(reset_ack(Bin,ReqId,Token),Bin), Bin).

bin_inI(reset_bin(Bin,ReqId,Token),control_center) :>
    shared_token(Token),
    agent(Bin),
    \+ last_reset_req(ReqId),
    assert(last_reset_req(ReqId)),
    evi(apply_reset_now),
    messageA(control_center, send_message(reset_ack(Bin,ReqId,Token),Bin), Bin).

bin_inI(retry_collection,_From) :>
    evi(retry_collection).

bin_inI(retry_collection(Bin,ReqId,Token),control_center) :>
    shared_token(Token),
    agent(Bin),
    format('[SMART_BIN ~w] RETRY requested for request=~w~n',[Bin,ReqId]),
    evi(retry_collection).

bin_inI(inform(reset_bin,_),_From) :>
    evi(reset_bin).

bin_inI(inform(retry_collection,_),_From) :>
    evi(retry_collection).

monitor(dummy) :-
    evi(monitor(dummy)).
