:- dynamic truck_state/1, current_job/2, started/0, status_counter/1, assign_wait_counter/1, seen_pickup/1, seen_assignment/1, completed_local/1, last_monitor_ts/1, busy_logged_req/1, move_counter/1, collect_counter/1.

deltaT(4).
move_time(2).
collect_time(1).
assign_wait_limit(6).
shared_token(city_token_2026).
monitor_interval_ms(1000).

major_truck_action(pickup_accept).
major_truck_action(move_start).
major_truck_action(collect_done).

truck_printI(Action,Bin) :>
    major_truck_action(Action),
    agent(T),
    atom_concat(truck,N,T),
    catch(atom_number(N,Num),_,fail),
    !,
    Delay is (Num - 1) * 0.25,
    sleep(Delay),
    ( atom_concat(smart_bin,Nb,Bin) -> atom_concat(smartbin,Nb,Bp) ; Bp = Bin ),
    format('[truck ~w] ~w at ~w~n',[T,Action,Bp]).

truck_printI(_,_) :>
    true.

truck_print_etaI(Action,Bin,EtaSec) :>
    major_truck_action(Action),
    agent(T),
    atom_concat(truck,N,T),
    catch(atom_number(N,Num),_,fail),
    !,
    Delay is (Num - 1) * 0.25,
    sleep(Delay),
    ( atom_concat(smart_bin,Nb,Bin) -> atom_concat(smartbin,Nb,Bp) ; Bp = Bin ),
    format('[truck ~w] ~w to ~w eta=~ws~n',[T,Action,Bp,EtaSec]).

truck_print_etaI(_,_,_) :>
    true.

startI :>
    retractall(started),
    assert(started),
    retractall(current_job(_,_)),
    retractall(truck_state(_)),
    retractall(assign_wait_counter(_)),
    retractall(seen_pickup(_)),
    retractall(seen_assignment(_)),
    retractall(completed_local(_)),
    retractall(busy_logged_req(_)),
    retractall(move_counter(_)),
    retractall(collect_counter(_)),
    retractall(last_monitor_ts(_)),
    assert(truck_state(idle)),
    deltaT(D),
    retractall(status_counter(_)),
    assert(status_counter(D)),
    statistics(walltime,[Now,_]),
    assert(last_monitor_ts(Now)).

status_checkI :>
    true.

send_message(send_message(Msg,_),From) :-
    !,
    send_message(Msg,From).

send_message(inform(Msg,_),From) :-
    !,
    send_message(Msg,From).

send_message(Msg,From) :-
    agent(T),
    format('[truck ~w] RX from=~w msg=~w~n',[T,From,Msg]),
    evi(truck_in(Msg,From)).

truck_inI(request(pickup(Bin),_),_From) :>
    evi(pickup_request(Bin,0)).

truck_inI(pickup_request(Bin),_From) :>
    evi(pickup_request(Bin,0)).

truck_inI(pickup_request(Bin,ReqId,Token),control_center) :>
    shared_token(Token),
        evi(pickup_request(Bin,ReqId)).

truck_inI(confirm(assignment(Bin),_),_From) :>
    evi(assignment(Bin,0)).

truck_inI(assignment(Bin),_From) :>
    evi(assignment(Bin,0)).

truck_inI(assignment(Bin,ReqId,Token),control_center) :>
    shared_token(Token),
        evi(assignment(Bin,ReqId)).

truck_inI(assignment(Bin,ReqId),control_center) :>
        evi(assignment(Bin,ReqId)).

pickup_requestE(Bin,ReqId) :>
    ReqId > 0,
    completed_local(ReqId),
    !,
    true.

pickup_requestE(Bin,ReqId) :>
    ReqId > 0,
    seen_pickup(ReqId),
    \+ current_job(Bin,ReqId),
    !,
    true.

pickup_requestE(Bin,ReqId) :>
    seen_pickup(ReqId),
    current_job(Bin,ReqId),
    truck_state(waiting_assignment),
    ReqId > 0,
    !,
    agent(T),
    shared_token(Token),
    evi(send_accept(Bin,ReqId)).

pickup_requestE(Bin,ReqId) :>
    truck_state(S),
    S \= idle,
    ReqId > 0,
    \+ busy_logged_req(ReqId),
    assert(busy_logged_req(ReqId)),
    !,
    agent(T),
    evi(send_refuse(Bin,ReqId)).

pickup_requestE(Bin,ReqId) :>
    truck_state(S),
    S \= idle,
    !,
    agent(T),
    evi(send_refuse(Bin,ReqId)).

pickup_requestE(Bin,ReqId) :>
    truck_state(idle),
    evi(truck_print(pickup_accept,Bin)),
    retract(truck_state(idle)),
    assert(truck_state(moving)),
    retractall(current_job(_,_)),
    assert(current_job(Bin,ReqId)),
    move_time(M),
    retractall(move_counter(_)),
    assert(move_counter(M)),
    evi(truck_print_eta(move_start,Bin,M)),
    evi(mark_seen_pickup(ReqId)),
    evi(send_accept(Bin,ReqId)).

assignmentE(Bin,ReqId) :>
    ReqId > 0,
    completed_local(ReqId),
    !,
    true.

assignmentE(Bin,ReqId) :>
    ReqId > 0,
    seen_assignment(ReqId),
    current_job(Bin,ReqId),
    !,
    evi(mark_assignment_ack(Bin,ReqId)).

assignmentE(Bin,ReqId) :>
    current_job(Bin,ReqId),
    !,
    agent(T),
    retractall(assign_wait_counter(_)),
    retractall(truck_state(_)),
    assert(truck_state(moving)),
    move_time(M),
    retractall(move_counter(_)),
    assert(move_counter(M)),
    evi(truck_print_eta(move_start,Bin,M)),
    evi(mark_assignment_ack(Bin,ReqId)),
    true.

assignmentE(_,_) :> true.

decide_outcomeI(Bin,ReqId) :>
    random(0,100,R),
    evi(decide_with_roll(R,Bin,ReqId)).

decide_with_rollI(_R,Bin,ReqId) :>
    evi(success(Bin,ReqId)).

successI(Bin,ReqId) :>
    evi(truck_print(collect_done,Bin)),
    evi(send_complete(Bin,ReqId)),
    evi(mark_completed(ReqId)),
    evi(cleanup).

cleanupI :>
    retractall(current_job(_,_)),
    retractall(assign_wait_counter(_)),
    retractall(move_counter(_)),
    retractall(collect_counter(_)),
    retractall(truck_state(_)),
    assert(truck_state(idle)).

assignment_timeoutI :>
    truck_state(waiting_assignment),
    current_job(Bin,ReqId),
    evi(send_refuse(Bin,ReqId)),
    evi(cleanup).

mark_seen_pickupI(ReqId) :>
    ReqId > 0,
    assert(seen_pickup(ReqId)).

mark_seen_pickupI(ReqId) :>
    ReqId =< 0.

mark_assignment_ackI(Bin,ReqId) :>
    ReqId > 0,
    agent(T),
    assert(seen_assignment(ReqId)),
    shared_token(Token),
    messageA(control_center, send_message(assignment_ack(T,Bin,ReqId,Token),T), T),
    messageA(control_center, send_message(assignment_ack(T,Bin,ReqId),T), T).

mark_assignment_ackI(_Bin,ReqId) :>
    ReqId =< 0.

mark_completedI(ReqId) :>
    ReqId > 0,
    assert(completed_local(ReqId)).

mark_completedI(ReqId) :>
    ReqId =< 0.

send_refuseI(Bin,ReqId) :>
    ReqId > 0,
    agent(T),
    shared_token(Token),
    messageA(control_center, send_message(job_refuse(T,Bin,ReqId,Token),T), T).

send_refuseI(Bin,ReqId) :>
    ReqId =< 0,
    agent(T),
    messageA(control_center, send_message(job_refuse(T,Bin),T), T).

send_acceptI(Bin,ReqId) :>
    ReqId > 0,
    agent(T),
    shared_token(Token),
    messageA(control_center, send_message(job_accept(T,Bin,ReqId,Token),T), T).

send_acceptI(Bin,ReqId) :>
    ReqId =< 0,
    agent(T),
    messageA(control_center, send_message(job_accept(T,Bin),T), T).

send_completeI(Bin,ReqId) :>
    ReqId > 0,
    evi(send_complete_token(Bin,ReqId)),
    evi(send_complete_req(Bin,ReqId)),
    evi(send_complete_legacy(Bin)).

send_complete_tokenI(Bin,ReqId) :>
    agent(T),
    shared_token(Token),
    messageA(control_center, send_message(collection_complete(Bin,ReqId,Token),T), T).

send_complete_reqI(Bin,ReqId) :>
    agent(T),
    messageA(control_center, send_message(collection_complete(Bin,ReqId),T), T).

send_complete_legacyI(Bin) :>
    agent(T),
    messageA(control_center, send_message(collection_complete(Bin),T), T).

send_completeI(Bin,ReqId) :>
    ReqId =< 0,
    agent(T),
    messageA(control_center, send_message(collection_complete(Bin),T), T).

monitorI(dummy) :>
    \+ started,
    evi(start).

monitorI(dummy) :>
    started,
    statistics(walltime,[Now,_]),
    last_monitor_ts(Last),
    monitor_interval_ms(Interval),
    Now - Last < Interval,
    !,
    true.

monitorI(dummy) :>
    started,
    statistics(walltime,[Now,_]),
    last_monitor_ts(Last),
    monitor_interval_ms(Interval),
    Now - Last >= Interval,
    retract(last_monitor_ts(Last)),
    assert(last_monitor_ts(Now)),
    evi(monitor_tick).

monitor_tickI :>
    evi(tick_status_counter),
    evi(tick_assignment_wait),
    evi(tick_move_phase),
    evi(tick_collect_phase).

tick_status_counterI :>
    status_counter(C),
    C > 1,
    C1 is C - 1,
    retract(status_counter(C)),
    assert(status_counter(C1)),
    !.

tick_status_counterI :>
    status_counter(1),
    deltaT(D),
    retract(status_counter(1)),
    assert(status_counter(D)),
    evi(status_check),
    !.
tick_status_counterI :> true.

tick_assignment_waitI :>
    truck_state(waiting_assignment),
    assign_wait_counter(C),
    C > 1,
    C1 is C - 1,
    retract(assign_wait_counter(C)),
    assert(assign_wait_counter(C1)),
    !.

tick_assignment_waitI :>
    truck_state(waiting_assignment),
    assign_wait_counter(1),
    retract(assign_wait_counter(1)),
    evi(assignment_timeout),
    !.
tick_assignment_waitI :> true.

tick_move_phaseI :>
    truck_state(moving),
    move_counter(C),
    C > 1,
    C1 is C - 1,
    retract(move_counter(C)),
    assert(move_counter(C1)),
    !.

tick_move_phaseI :>
    truck_state(moving),
    move_counter(1),
    current_job(Bin,ReqId),
    retract(move_counter(1)),
    collect_time(Ct),
    retractall(collect_counter(_)),
    assert(collect_counter(Ct)),
    retract(truck_state(moving)),
    assert(truck_state(collecting)),
    evi(truck_print_eta(collect_start,Bin,Ct)),
    !.
tick_move_phaseI :> true.

tick_collect_phaseI :>
    truck_state(collecting),
    collect_counter(C),
    C > 1,
    C1 is C - 1,
    retract(collect_counter(C)),
    assert(collect_counter(C1)),
    !.

tick_collect_phaseI :>
    truck_state(collecting),
    collect_counter(1),
    current_job(Bin,ReqId),
    retract(collect_counter(1)),
    decide_outcomeI(Bin,ReqId),
    !.
tick_collect_phaseI :> true.

monitor(dummy) :-
    evi(monitor(dummy)).
